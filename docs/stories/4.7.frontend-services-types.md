### Data Models
**Complete Hierarchy Types**:
```typescript
// Base hierarchy interface
interface HierarchyItem {
  id: string;
  title: string;
  description: string | null;
  status: HierarchyStatus;
  created_at: string;
  updated_at: string;
}

type HierarchyStatus = 'todo' | 'doing' | 'review' | 'waiting' | 'done';
type Priority = 'low' | 'medium' | 'high' | 'critical';

interface Epic extends HierarchyItem {
  project_id: string;
  priority: Priority;
  mvp_flag: boolean;
  progress?: number;
  stories?: Story[];
}

interface Story extends HierarchyItem {
  epic_id: string;
  priority: Priority;
  mvp_flag: boolean;
  progress?: number;
  tasks?: Task[];
}

interface Task extends HierarchyItem {
  story_id: string;
  parent_task_id?: string; // For Subtasks
  assignee: 'User' | 'Archon' | 'AI IDE Agent';
  task_order: number;
  progress?: number;
  subtasks?: Task[];
}

// Service operation types
interface CreateEpicRequest {
  project_id: string;
  title: string;
  description?: string;
  priority?: Priority;
  mvp_flag?: boolean;
}

interface UpdateEpicRequest {
  title?: string;
  description?: string;
  status?: HierarchyStatus;
  priority?: Priority;
  mvp_flag?: boolean;
}
```

### React Query Integration
**Hook Patterns**:

// Epic Query Hooks
export const useEpics = (projectId: string) =>
  useQuery(['epics', projectId], () => epicService.listEpics(projectId));

export const useEpic = (epicId: string) =>
  useQuery(['epic', epicId], () => epicService.getEpic(epicId));

export const useCreateEpic = () =>
  useMutation(epicService.createEpic, {
    onSuccess: () => queryClient.invalidateQueries(['epics'])
