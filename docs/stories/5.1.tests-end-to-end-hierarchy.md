```typescript
test('Create complete hierarchy from Project to Subtask', async ({ page }) => {
  // Create EPIC
  const epic = await createEpic(page, {
    title: 'Test Epic',
    code: 'E-01',
    status: 'todo',
    priority: 'high',
    mvp_flag: true
  });

  // Create STORY under EPIC
  const story = await createStory(page, {
    epic_id: epic.id,
    title: 'Test Story',
    code: 'S-01-01',
    status: 'todo'
  });

  // Create TASK under STORY
  const task = await createTask(page, {
    story_id: story.id,
    title: 'Test Task',
    code: 'T-01-01-01',
    status: 'todo'
  });

  // Create SUBTASK under TASK
  const subtask = await createSubtask(page, {
    parent_task_id: task.id,
    title: 'Test Subtask',
    status: 'todo'
  });

  // Validate hierarchy display
  await validateHierarchyView(page, epic, story, task, subtask);
});
```

#### Scenario 2: Dependency Cycle Detection
```typescript
test('Detect and prevent dependency cycles', async ({ page }) => {
  // Create 3 tasks
  const task1 = await createTask(page, { title: 'Task 1' });
  const task2 = await createTask(page, { title: 'Task 2' });
  const task3 = await createTask(page, { title: 'Task 3' });

  // Create dependencies: task1 → task2 → task3
  await createDependency(page, task1.id, task2.id, 'blocks');
  await createDependency(page, task2.id, task3.id, 'blocks');

  // Try to create cycle: task3 → task1 (should fail)
  const result = await createDependency(page, task3.id, task1.id, 'blocks');

  expect(result.error).toContain('cycle detected');
});
```

#### Scenario 3: Performance Under Load
```typescript
test('Handle 1000+ items with acceptable performance', async ({ page }) => {

  // Test response times
```
